%================= EXACT LEARNABILITY PART I: NTK PREDICTOR =================
\section{Exact Learnability Part I: NTK Predictor}

\begin{frame}{Bước 2: Orthonormalized Training (Section 5.1)}
  \textbf{Mã hóa Template thành Training Data:}
  \begin{itemize}
    \item Mỗi template $(x^{(i,j)}, y^{(i,j)}) \in T_i$ được mã hóa thành vector $q_{ij} \in \mathbb{R}^{k'}$.
    \item $q_{ij}$ có cấu trúc block-partitioned: chỉ block thứ $i$ khác 0.
    \item Các $q_{ij}$ được \textbf{trực giao hóa} (orthogonalized) trong mỗi block.
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Kết quả:}
  \[
    X = \{q_{ij} : i \in [b], j \in [t_i]\} \subseteq \mathbb{R}^{k'} \quad \text{(orthonormal)}
  \]
  
  $\Rightarrow$ NTK matrix $\Theta(X,X)$ trở thành \textbf{scaled identity + rank-1 noise}.
\end{frame}

\begin{frame}{Test Input Structure}
  \textbf{Cấu trúc Test Input $\hat{x}$:}
  \begin{itemize}
    \item $\hat{x} = \frac{1}{\sqrt{n_{\hat{x}}}} (\hat{x}_1, \ldots, \hat{x}_b)^\top$
    \item Mỗi block $\hat{x}_i$ hoặc = $0$ hoặc match một sample trong $X$.
    \item $n_{\hat{x}}$ = số blocks active (matched).
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Quan trọng:}
  \begin{itemize}
    \item Tổng số test inputs: $O(2^b)$ (exponentially lớn hơn training set $O(b)$).
    \item Ví dụ: Addition 10-bit cần 40 training samples nhưng có $4^{10}$ test inputs.
  \end{itemize}
\end{frame}

\begin{frame}{Theorem 5.1: NTK Predictor Behavior}
  \textbf{Assumption 5.1 (Bounded Unwanted Correlations):}
  Số "unwanted correlations" (training samples không match $\hat{x}$ nhưng có bit $i$ set) phải nhỏ hơn tỷ lệ $-w_1/w_0$.

  \vspace{0.3cm}
  \textbf{Theorem 5.1:} Dưới Assumption 5.1, NTK predictor $\mu(\hat{x})$ bảo toàn dấu:
  \[
    \begin{cases}
      \mu(\hat{x})_i \le 0 & \text{nếu ground-truth bit } f(\hat{x})_i = 0 \\
      \mu(\hat{x})_i > 0 & \text{nếu ground-truth bit } f(\hat{x})_i = 1
    \end{cases}
  \]

  $\Rightarrow$ \textbf{Rounding} cho kết quả chính xác:
  \[
    \text{round}(\mu(\hat{x})_i) = f(\hat{x})_i
  \]
\end{frame}

%================= EXACT LEARNABILITY PART II: ENSEMBLE COMPLEXITY =================
\section{Exact Learnability Part II: Ensemble Complexity}

\begin{frame}{Bước 4: Ensemble Complexity Bound (Section 6)}
  \textbf{Vấn đề:} Theorem 5.1 chỉ đúng cho \emph{kỳ vọng} $\mu(\hat{x})$. Thực tế mỗi model có variance $\sigma^2(\hat{x})$.

  \vspace{0.3cm}
  \textbf{Giải pháp:} Ensemble $N$ models, lấy trung bình $G(\hat{x}) = \frac{1}{N}\sum_{j=1}^N F^j(\hat{x})$.

  \vspace{0.3cm}
  \textbf{Công thức Ensemble Complexity (Equation 8):}
  \[
    N \ge 8 \max_{\hat{x}, i \in [k]} \left\{ \frac{\sigma^2(\hat{x})}{\mu^2(\hat{x})_i} \right\} \ln\left(\frac{2k'}{\delta}\right)
  \]

  Với xác suất $1 - \delta$, rounding ensemble mean cho kết quả chính xác.
\end{frame}

\begin{frame}{Lemma 6.1 và Remark 6.1: Asymptotic Order}
  \textbf{Lemma 6.1:} Phân tích bậc của $\mu(\hat{x})$ và $\sigma^2(\hat{x})$:
  \begin{itemize}
    \item $\sigma^2(\hat{x}) \in O(1/k')$
    \item $|\mu(\hat{x})_i|$ phụ thuộc vào $n_{\hat{x}}$ và $k'$.
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Remark 6.1 (Kết quả chính):}
  \begin{itemize}
    \item \textbf{Permutation:} $N \in O(\ell^2)$
    \item \textbf{Addition:} $N \in O(\ell^2)$
    \item \textbf{Multiplication:} $N \in O(\ell^2)$
  \end{itemize}

  Trong đó $\ell$ = số bit của input.
\end{frame}

\section{Kết quả thực nghiệm}

\begin{frame}{Validation thực nghiệm (Paper Figure 5)}
  \textbf{Setup thí nghiệm:}
  \begin{itemize}
    \item Mạng 2 lớp, 50,000 hidden units.
    \item Full-batch gradient descent.
    \item Bit length $\ell$ từ 2 đến 30.
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Kết quả:}
  \begin{itemize}
    \item Ensemble complexity tăng theo $O(\ell^2 \log \ell)$ như lý thuyết dự đoán.
    \item Numerical (xanh dương) khớp hoàn hảo với Theoretical bound (xanh lá).
  \end{itemize}
\end{frame}
